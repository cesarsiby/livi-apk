
r.post('/intercity/guarantees/:id/confirm-funded', requireRoles('admin'), asyncHandler(async(req,res)=>{const b=z.object({funding_reference:z.string().min(2).max(160)}).parse(req.body||{});const result=await tx(async c=>{const g=(await c.query(`SELECT * FROM intercity_guarantees WHERE id=$1 FOR UPDATE`,[req.params.id])).rows[0];if(!g)throw new HttpError(404,'Garantie interville introuvable');if(g.status==='held')return {id:g.id,status:g.status,idempotent:true};if(g.status!=='awaiting_funding')throw new HttpError(409,'Garantie non éligible à la confirmation de financement.','INTERCITY_GUARANTEE_STATE');return (await c.query(`UPDATE intercity_guarantees SET status='held',funding_reference=$2,funded_at=now(),updated_at=now() WHERE id=$1 RETURNING *`,[g.id,b.funding_reference])).rows[0];});ok(res,result)}));
r.post('/intercity/guarantees/:id/confirm-released', requireRoles('admin'), asyncHandler(async(req,res)=>{const b=z.object({release_reference:z.string().min(2).max(160)}).parse(req.body||{});const result=await tx(async c=>{const g=(await c.query(`SELECT * FROM intercity_guarantees WHERE id=$1 FOR UPDATE`,[req.params.id])).rows[0];if(!g)throw new HttpError(404,'Garantie interville introuvable');if(g.status==='released_to_partner')return {id:g.id,status:g.status,idempotent:true};if(g.status!=='release_pending')throw new HttpError(409,'Garantie non éligible à la libération partenaire.','INTERCITY_GUARANTEE_STATE');return (await c.query(`UPDATE intercity_guarantees SET status='released_to_partner',release_reference=$2,released_at=now(),updated_at=now() WHERE id=$1 RETURNING *`,[g.id,b.release_reference])).rows[0];});ok(res,result)}));
r.post('/intercity/guarantees/:id/pay-claim', requireRoles('admin'), asyncHandler(async(req,res)=>{const b=z.object({settlement_reference:z.string().min(2).max(160),amount_xof:z.number().int().positive()}).parse(req.body||{});const result=await tx(async c=>{const g=(await c.query(`SELECT g.*,o.buyer_id FROM intercity_guarantees g JOIN orders o ON o.id=g.order_id WHERE g.id=$1 FOR UPDATE`,[req.params.id])).rows[0];if(!g)throw new HttpError(404,'Garantie interville introuvable');if(g.status==='paid_to_buyer')return {id:g.id,status:g.status,idempotent:true};if(g.status!=='claim_pending')throw new HttpError(409,'Garantie non éligible au paiement de sinistre.','INTERCITY_GUARANTEE_STATE');if(BigInt(b.amount_xof)!==BigInt(g.guarantee_amount_xof))throw new HttpError(422,'Le montant du sinistre doit correspondre à la garantie contractuelle.','INTERCITY_GUARANTEE_AMOUNT_MISMATCH');const out=(await c.query(`UPDATE intercity_guarantees SET status='paid_to_buyer',claim_reference=$2,paid_to_buyer_at=now(),updated_at=now() WHERE id=$1 RETURNING *`,[g.id,b.claim_reference])).rows[0];return out;});ok(res,result)}));
